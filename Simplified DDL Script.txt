CREATE DATABASE IF NOT EXISTS `banana_go`;
USE `banana_go`;

CREATE TABLE `users` (
 `id_user` int(11) NOT NULL AUTO_INCREMENT,
 `username` varchar(100) NOT NULL,
 `password` varchar(255) NOT NULL,
 `role` enum('admin','user') NOT NULL DEFAULT 'user',
 PRIMARY KEY (`id_user`),
 UNIQUE KEY `username` (`username`)
)

CREATE TABLE `messages` (
  `id_message` int(11) NOT NULL AUTO_INCREMENT,
  `nama` varchar(255),
  `email` varchar(255),
  `pesan` varchar(255),
  `created_at` datetime DEFAULT NULL,
  PRIMARY KEY (`id_message`)
) 

CREATE TABLE `products` (
  `id_product` int(11) NOT NULL AUTO_INCREMENT,
  `nama_product` varchar(150) NOT NULL,
  `harga` int(11) NOT NULL,
  `gambar` varchar(255) NOT NULL,
  `stok` int(11) NOT NULL DEFAULT 0,
  `created_at` timestamp NOT NULL DEFAULT current_timestamp(),
  `categories` enum('Cheese','Chocolate','Matcha','Mix','Drinks','Sop Buah') DEFAULT NULL,
  PRIMARY KEY (`id_product`)
) 

CREATE TABLE `orders` (
  `id_order` int(11) NOT NULL AUTO_INCREMENT,
  `id_user` int(11) NOT NULL,
  `catatan` text DEFAULT NULL,
  `status` enum('pending','dikonfirmasi','ditolak') DEFAULT 'pending',
  `created_at` timestamp NOT NULL DEFAULT current_timestamp(),
  `stock_dikurangi` enum('belum','sudah') DEFAULT 'belum',
  PRIMARY KEY (`id_order`),
  KEY `id_user` (`id_user`),
  CONSTRAINT `orders_ibfk_1` FOREIGN KEY (`id_user`) REFERENCES `users` (`id_user`) ON DELETE CASCADE ON UPDATE CASCADE
)

CREATE TABLE `order_detail` (
  `id_detail` int(11) NOT NULL AUTO_INCREMENT,
  `id_order` int(11) NOT NULL,
  `id_product` int(11) NOT NULL,
  `jumlah` int(11) NOT NULL,
  `harga` int(11) NOT NULL,
  PRIMARY KEY (`id_detail`),
  KEY `id_order` (`id_order`),
  KEY `id_product` (`id_product`),
  CONSTRAINT `order_detail_ibfk_1` FOREIGN KEY (`id_order`) REFERENCES `orders` (`id_order`) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT `order_detail_ibfk_2` FOREIGN KEY (`id_product`) REFERENCES `products` (`id_product`) ON DELETE CASCADE ON UPDATE CASCADE
)
